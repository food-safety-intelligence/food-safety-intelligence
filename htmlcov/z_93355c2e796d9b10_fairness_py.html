1"""Disparity metrics engine — city-agnostic, reads only the ``AuditFrame``. 

2 

3For one grouping axis this computes, per group: flag rate, false-positive rate, 

4false-negative rate, and calibration error, then the across-group gaps, each with 

5a bootstrap confidence interval and a verdict against the tolerance bands in 

6``config``. A gap is only called a *finding* when it is both **material** (point 

7estimate past the band) and **confident** (the CI stays past the band) — small, 

8low-prevalence groups otherwise trip false alarms on base-rate artifacts alone 

9(decision 0005). 

10 

11"Flagged" = the deployed risk tier(s) in ``config.FLAGGED_TIERS_PRIMARY`` (High). 

12Everything here is measurement only — no census column is ever a model input. 

13"""

35def _ece(y: np.ndarray, score: np.ndarray, *, n_bins: int = 10) -> float: 

36 """Expected calibration error: weighted mean |mean predicted - mean observed| 

37 over ``n_bins`` equal-width score bins. NaN for an empty slice.""" 

38 if len(y) == 0: 38 ↛ 39line 38 didn't jump to line 39 because the condition on line 38 was never true

39 return float("nan") 

40 idx = np.clip((score * n_bins).astype(int), 0, n_bins - 1) 

41 total = 0.0 

42 for b in range(n_bins): 

43 m = idx == b 

44 if m.any(): 

45 total += m.mean() * abs(score[m].mean() - y[m].mean()) 

46 return float(total)

49def _range(vals: np.ndarray) -> float: 

50 """max - min ignoring NaN; NaN if fewer than two groups have a value.""" 

51 v = vals[~np.isnan(vals)] 

52 return float(v.max() - v.min()) if v.size >= 2 else float("nan")

55def _gaps_from_arrays( 

56 y: np.ndarray, 

57 flagged: np.ndarray, 

58 score: np.ndarray, 

59 g: np.ndarray, 

60 n_groups: int, 

61 *, 

62 min_pos: int = config.MIN_GROUP_POSITIVES, 

63) -> dict[str, float]: 

64 """The four across-group gaps for one axis, from raw arrays (bootstrap-hot). 

65 

66 Flag rate / FPR / disparate-impact ratio only need the row-count floor (applied 

67 upstream when selecting audited groups). FNR and calibration condition on the 

68 positives, so a group with fewer than ``min_pos`` positives is masked (NaN) for 

69 those two metrics — it still counts for parity. 

70 """ 

71 flag_rate = np.full(n_groups, np.nan) 

72 fpr = np.full(n_groups, np.nan) 

73 fnr = np.full(n_groups, np.nan) 

74 ece = np.full(n_groups, np.nan) 

75 for k in range(n_groups): 

76 mk = g == k 

77 if not mk.any(): 77 ↛ 78line 77 didn't jump to line 78 because the condition on line 77 was never true

78 continue 

79 yk, flk, sck = y[mk], flagged[mk], score[mk] 

80 flag_rate[k] = flk.mean() 

81 neg, pos = yk == 0, yk == 1 

82 if neg.any(): 82 ↛ 84line 82 didn't jump to line 84 because the condition on line 82 was always true

83 fpr[k] = flk[neg].mean() 

84 if int(pos.sum()) >= min_pos: # FNR / calibration need enough positives 

85 fnr[k] = 1.0 - flk[pos].mean() 

86 ece[k] = _ece(yk, sck) 

87 valid_fr = flag_rate[~np.isnan(flag_rate)] 

88 top = valid_fr.max() if valid_fr.size else np.nan 

89 di_ratio = (valid_fr.min() / top) if (valid_fr.size >= 2 and top > 0) else float("nan") 

90 return { 

91 "disparate_impact_ratio": di_ratio, 

92 "fpr_gap": _range(fpr), 

93 "fnr_gap": _range(fnr), 

94 "ece_gap": _range(ece), 

95 }

114 @property 

115 def any_finding(self) -> bool: 

116 return bool(self.gaps["finding"].any()) if len(self.gaps) else False

119# Tolerance band + verdict direction per gap metric. 

120# "min": worse when smaller (disparate-impact ratio) -> finding if value < band 

121# AND ci_high < band. 

122# "max": worse when larger (rate/error gaps) -> finding if value > band 

123# AND ci_low > band. 

124_GAP_SPEC: dict[str, tuple[float, str]] = { 

125 "disparate_impact_ratio": (config.DISPARATE_IMPACT_RATIO_MIN, "min"), 

126 "fpr_gap": (config.FPR_GAP_MAX, "max"), 

127 "fnr_gap": (config.FNR_GAP_MAX, "max"), 

128 "ece_gap": (config.ECE_GAP_MAX, "max"), 

129}

132def group_table( 

133 frame: pd.DataFrame, column: str, *, flagged_tiers=config.FLAGGED_TIERS_PRIMARY 

134) -> pd.DataFrame: 

135 """Per-group metrics for every above-floor group of ``column``. 

136 

137 Rows with a missing group value are excluded (reported via coverage, never 

138 imputed). A group is *audited* only when it clears both noise floors. 

139 """ 

140 df = frame[[column, "y_true", "risk_tier", "y_score"]].copy() 

141 df = df[df[column].notna()] 

142 df["_flagged"] = flagged_mask(df["risk_tier"], flagged_tiers).astype(int) 

143 

144 rows = [] 

145 for name, grp in df.groupby(column, observed=True): 

146 y = grp["y_true"].to_numpy() 

147 fl = grp["_flagged"].to_numpy() 

148 sc = grp["y_score"].to_numpy() 

149 pos = int((y == 1).sum()) 

150 # Parity/FPR audit on the row-count floor; FNR/calibration also need positives. 

151 audited = len(grp) >= config.MIN_GROUP_N 

152 odds_reliable = audited and pos >= config.MIN_GROUP_POSITIVES 

153 rows.append( 

154 { 

155 "group": name, 

156 "n": int(len(grp)), 

157 "positives": pos, 

158 "prevalence": round(float(y.mean()), 4), 

159 "flag_rate": round(float(fl.mean()), 4), 

160 "fpr": round(float(fl[y == 0].mean()), 4) if (y == 0).any() else float("nan"), 

161 "fnr": round(1.0 - float(fl[y == 1].mean()), 4) if odds_reliable else float("nan"), 

162 "ece": round(_ece(y, sc), 4) if odds_reliable else float("nan"), 

163 "mean_pred": round(float(sc.mean()), 4), 

164 "mean_obs": round(float(y.mean()), 4), 

165 "audited": audited, 

166 "odds_reliable": odds_reliable, 

167 } 

168 ) 

169 out = pd.DataFrame(rows).sort_values("n", ascending=False).reset_index(drop=True) 

170 return out

173def audit_axis( 

174 frame: pd.DataFrame, 

175 axis: config.Axis, 

176 *, 

177 flagged_tiers=config.FLAGGED_TIERS_PRIMARY, 

178 n_bootstrap: int = config.N_BOOTSTRAP, 

179 seed: int = RANDOM_STATE, 

180) -> AxisResult: 

181 """Full disparity audit for one axis: group table + gap CIs + verdicts.""" 

182 column = axis.column 

183 gt = group_table(frame, column, flagged_tiers=flagged_tiers) 

184 audited = gt[gt["audited"]]["group"].tolist() 

185 

186 # Restrict to audited groups and factorise for the bootstrap-hot arrays. 

187 sub = frame[frame[column].isin(audited)] 

188 coverage = float(len(sub) / len(frame)) if len(frame) else 0.0 

189 g, _ = pd.factorize(sub[column]) 

190 n_groups = len(audited) 

191 y = sub["y_true"].to_numpy() 

192 fl = flagged_mask(sub["risk_tier"], flagged_tiers).to_numpy().astype(int) 

193 sc = sub["y_score"].to_numpy() 

194 

195 if n_groups < 2: 

196 gaps = pd.DataFrame( 

197 columns=["metric", "value", "ci_low", "ci_high", "tolerance", "direction", "finding"] 

198 ) 

199 return AxisResult(axis.key, column, gt, gaps, n_groups, coverage) 

200 

201 point = _gaps_from_arrays(y, fl, sc, g, n_groups) 

202 

203 # Bootstrap the gaps by resampling rows with replacement. 

204 rng = np.random.default_rng(seed) 

205 n = len(y) 

206 boot: dict[str, list[float]] = {m: [] for m in _GAP_SPEC} 

207 for _ in range(n_bootstrap): 

208 idx = rng.integers(0, n, n) 

209 b = _gaps_from_arrays(y[idx], fl[idx], sc[idx], g[idx], n_groups) 

210 for m in _GAP_SPEC: 

211 boot[m].append(b[m]) 

212 

213 rows = [] 

214 for metric, (band, direction) in _GAP_SPEC.items(): 

215 arr = np.asarray(boot[metric], dtype=float) 

216 arr = arr[~np.isnan(arr)] 

217 lo, hi = np.percentile(arr, [2.5, 97.5]) if arr.size else (np.nan, np.nan) 

218 value = point[metric] 

219 if direction == "min": # disparate-impact ratio: confident below the band 

220 finding = bool(value < band and hi < band) 

221 else: # rate / error gap: confident above the band 

222 finding = bool(value > band and lo > band) 

223 rows.append( 

224 { 

225 "metric": metric, 

226 "value": round(float(value), 4) if not np.isnan(value) else float("nan"), 

227 "ci_low": round(float(lo), 4) if not np.isnan(lo) else float("nan"), 

228 "ci_high": round(float(hi), 4) if not np.isnan(hi) else float("nan"), 

229 "tolerance": band, 

230 "direction": direction, 

231 "finding": finding, 

232 } 

233 ) 

234 gaps = pd.DataFrame(rows) 

235 return AxisResult(axis.key, column, gt, gaps, n_groups, coverage)

358 for idx in (rng.integers(0, n, n) for _ in range(n_bootstrap)) 

359 ] 

360 arr = np.asarray(boot, dtype=float) 

361 arr = arr[~np.isnan(arr)] 

362 lo, hi = np.percentile(arr, [2.5, 97.5]) if arr.size else (np.nan, np.nan)

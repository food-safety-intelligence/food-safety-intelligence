1""" 

2Shared scores.json matching primitives — Food Safety Intelligence agent. 

3------------------------------------------------------------------------ 

4The rules for deciding whether an OSM `name` / free-text name and a city 

5`dba_name` denote the same establishment, and for normalising addresses and 

6names before comparing, are subtle (store numbers, generic words, the 

7"Amarit" vs "AMARIT RESTAURANT" length mismatch). They must match identically 

8wherever the agent resolves a venue, so they live here once: 

9 

10 * get_safety_score — batch, OSM-driven (address + name, geo fallback) 

11 * look_up_establishment — name-only lookup for general chat 

12 

13These are PURE functions over records the caller has already loaded: no file 

14IO and no cache. Each tool keeps its own (cached) loader and passes the index 

15or record list in, so this module stays trivially testable and never reaches 

16for a data file. 

17"""

25# Address must be a close match AND the best name in that address bucket must 

26# clear its own bar before we attach a score. Name is the disambiguator at 

27# shared addresses, so it gets the stricter, independently-tuned cutoff. 

28ADDRESS_CUTOFF = 0.72 

29NAME_CUTOFF = 0.6 

30 

31# Name + geographic-proximity fallback, used only when the address match fails. 

32# OSM very often carries no clean street address (just "Chicago, IL" or nothing), 

33# so address-only matching reports a venue that IS in the batch run as having no 

34# record (e.g. "Amarit" -> the real "AMARIT RESTAURANT"). lat/lon + name recovers 

35# it: OSM and the city geocode the same building to slightly different points, so 

36# any record within ~0.0025° (≈250 m) is a candidate, then the name must agree. 

37# Proximity alone is not enough — a dense block holds many venues — so the name 

38# is what identifies the establishment (ethics decision record 0005, principle 1: 

39# a confident wrong score is worse than a miss). 

40GEO_RADIUS_DEG = 0.0025 

41# SequenceMatcher ratio bar for the near-identical / typo case; the token-subset 

42# rule below handles the common "Amarit" vs "AMARIT RESTAURANT" length mismatch 

43# that ratio alone scores too low (~0.55) to accept. 

44GEO_NAME_RATIO_CUTOFF = 0.87

106def normalise_address(addr: str) -> str: 

107 """Uppercase, reduce to the street portion, expand abbreviations, collapse space. 

108 

109 The two sides are written differently: OSM assembles a full postal address 

110 ("1115 North San Fernando Boulevard, Burbank, CA, 91504") while every city's 

111 scores.json stores a bare street line ("13736 AMAR RD") — measured at <0.05% 

112 commas across Chicago, NYC and LA. Comparing the two whole meant the address 

113 bucket essentially never hit outside Chicago, pushing every venue onto the 

114 ~250m geo fallback. Keeping only the text before the first comma puts both 

115 sides in the same shape; it is a no-op on the city records themselves. 

116 """ 

117 addr = addr.split(",")[0].upper() 

118 replacements = { 

119 "STREET": "ST", 

120 "AVENUE": "AVE", 

121 "BOULEVARD": "BLVD", 

122 "DRIVE": "DR", 

123 "COURT": "CT", 

124 "PLACE": "PL", 

125 "ROAD": "RD", 

126 "NORTH": "N", 

127 "SOUTH": "S", 

128 "EAST": "E", 

129 "WEST": "W", 

130 } 

131 for long, short in replacements.items(): 

132 addr = re.sub(rf"\b{long}\b", short, addr) 

133 return re.sub(r"\s+", " ", addr).strip()

136def normalise_name(name: str) -> str: 

137 """Uppercase, drop store numbers, strip punctuation, collapse whitespace. 

138 

139 OSM `name` and city `dba_name` are formatted very differently 

140 ("Dunkin'" vs "DUNKIN #305"), so fold both hard before comparing: remove 

141 "#1234"-style store numbers, turn any run of non-alphanumerics into a 

142 single space, and trim. 

143 """ 

144 # A scores.json record may carry an explicit null name; coerce so .upper() 

145 # never crashes on None. 

146 name = (name or "").upper() 

147 name = re.sub(r"#\s*\d+", " ", name) # store / franchise numbers 

148 name = name.replace("'", "").replace("’", "") # join contractions ("McDonald's") 

149 name = re.sub(r"[^A-Z0-9]+", " ", name) # remaining punctuation -> space 

150 return re.sub(r"\s+", " ", name).strip()

153def fuzzy_lookup(address: str, name: str, index: dict[str, list[dict]]) -> dict | None: 

154 """Return the best score record for this (address, name), or None. 

155 

156 Resolve the address to a bucket of records (exact key, then fuzzy over the 

157 keys), then pick the record in that bucket whose `dba_name` best matches 

158 `name`. BOTH the address and the best name must clear their cutoffs. 

159 

160 A shared address (food court, airport terminal) holds many establishments, 

161 so address alone can attach the wrong business's score — a consumer-facing 

162 wrong-signal harm (ethics decision record 0005, principle 1). A missed 

163 match (None) is safer than a confident wrong one. 

164 """ 

165 key = normalise_address(address) 

166 # Only a key carrying a house number pins a BUILDING. A bare street or landmark 

167 # ("BROADWAY", "PENN STATION", or the "LOS ANGELES" left when an OSM venue has no 

168 # street at all) names something miles long, so it is weak evidence however it 

169 # matched, and it must never be fuzzed: "LOS ANGELES" is a 0.72-similar string to 

170 # "435 LOS ANGELES ST" while naming nothing in common with it. 

171 key_pins_a_building = _has_house_number(key) 

172 bucket = index.get(key) 

173 address_is_exact = bucket is not None 

174 if bucket is None: 

175 if not key_pins_a_building: 

176 return None 

177 matches = difflib.get_close_matches(key, index.keys(), n=1, cutoff=ADDRESS_CUTOFF) 

178 if not matches: 

179 return None 

180 bucket = index[matches[0]] 

181 

182 # A single-occupancy address used to return its record with no name check at all, 

183 # which attached the wrong business's score in two ways, both seen on live Los 

184 # Angeles data: 

185 # * a fuzzy address hop lands in a different building's bucket — OSM 

186 # "Taco Bell, 1115 N San Fernando Blvd" resolved to "STARBUCKS COFFEE #9746"; 

187 # * the address matches exactly but the city record on it is a different tenant 

188 # — "Cafe Etc." resolved to "CAFFE HUB". 

189 # How hard to gate depends on how much the address already proved. An exact 

190 # house-numbered address is strong evidence, so only rule out names with nothing 

191 # in common; demanding a full match there would drop honest rewrites like OSM 

192 # "Amarit Thai" vs the city's "AMARIT RESTAURANT". Anything weaker — a fuzzy hop, 

193 # or an exact hit on a house-number-free key like "BROADWAY", which NYC publishes 

194 # 52 of — has to clear the full bar, or every venue on thirteen miles of Broadway 

195 # inherits one deli's score. 

196 if len(bucket) == 1: 

197 dba = bucket[0].get("dba_name", "") 

198 strong = address_is_exact and key_pins_a_building 

199 gate = shares_distinctive_token if strong else names_match 

200 return bucket[0] if gate(name, dba) else None 

201 

202 target = normalise_name(name) 

203 if not target: 

204 return None 

205 

206 best: dict | None = None 

207 best_ratio = 0.0 

208 for record in bucket: 

209 ratio = difflib.SequenceMatcher( 

210 None, target, normalise_name(record.get("dba_name", "")) 

211 ).ratio() 

212 # Strict `>` keeps the first record on an exact tie, so a tie resolves to 

213 # scores.json order. This only bites when two venues share both an 

214 # address and an identical name (near-indistinguishable), and we have no 

215 # better signal (no license_id) to break it — so first-in-order is fine. 

216 if ratio > best_ratio: 

217 best_ratio = ratio 

218 best = record 

219 

220 return best if best_ratio >= NAME_CUTOFF else None

223def _has_house_number(key: str) -> bool: 

224 """True if a normalised address key starts with a street number. 

225 

226 Both cities' records and OSM put the house number first ("13736 AMAR RD"), so a 

227 leading digit is what separates an address that pins one building from a bare 

228 street or landmark name. 

229 """ 

230 return bool(key) and key[0].isdigit()

233def shares_distinctive_token(osm_name: str, dba_name: str) -> bool: 

234 """True if the two names share at least one non-generic word. 

235 

236 A deliberately weaker bar than `names_match`, for the case where the ADDRESS 

237 already matched exactly and the only job is to catch a name that is obviously a 

238 different business. OSM and the city often write the same venue differently 

239 enough to fail `names_match` ("Amarit Thai" vs "AMARIT RESTAURANT" share no 

240 subset and score ~0.55), so demanding a full match there throws away real 

241 matches — but a genuinely different tenant shares no distinctive word at all 

242 ("Cafe Etc." vs "CAFFE HUB", "Taco Bell" vs "STARBUCKS COFFEE"). 

243 """ 

244 a = _distinctive_tokens(osm_name) 

245 b = _distinctive_tokens(dba_name) 

246 return bool(a & b)

249def _distinctive_tokens(name: str) -> set[str]: 

250 """Identifying words of a name: generic ones dropped, single letters excluded. 

251 

252 A one-character token carries no identity and collides constantly once 

253 punctuation is split ("L.A. Tacos" normalises to L / A / TACOS), so a shared "A" 

254 must never be enough to call two venues the same establishment. 

255 """ 

256 return {t for t in normalise_name(name).split() if len(t) > 1} - GENERIC_NAME_TOKENS

259def names_match(osm_name: str, dba_name: str) -> bool: 

260 """True if a free-text / OSM name and a city dba_name plausibly name the same venue. 

261 

262 Two complementary rules, because OSM and the city write names very 

263 differently: 

264 

265 * Token-subset: every word of the shorter name appears in the longer one, 

266 and the shorter name has at least one distinctive (non-generic) word. 

267 This accepts "Amarit" vs "AMARIT RESTAURANT" (a pure length mismatch that 

268 SequenceMatcher scores far too low) while rejecting "China Cafe" vs 

269 "China Grill" (neither is a subset of the other). 

270 * High SequenceMatcher ratio: catches near-identical spellings / word order 

271 that the subset rule misses. 

272 """ 

273 a = normalise_name(osm_name).split() 

274 b = normalise_name(dba_name).split() 

275 if not a or not b: 

276 return False 

277 

278 short, long = (a, b) if len(a) <= len(b) else (b, a) 

279 short_set, long_set = set(short), set(long) 

280 if short_set <= long_set and any(t not in GENERIC_NAME_TOKENS for t in short_set): 

281 return True 

282 

283 ratio = difflib.SequenceMatcher(None, " ".join(sorted(a)), " ".join(sorted(b))).ratio() 

284 return ratio >= GEO_NAME_RATIO_CUTOFF

287def geo_lookup(name: str, lat: float | None, lon: float | None, records: list[dict]) -> dict | None: 

288 """Find a score record by name within ~250 m of (lat, lon), or None. 

289 

290 The address-first path misses any venue whose OSM record lacks a clean 

291 street address; this recovers it from coordinates + name. Among all records 

292 inside the proximity box whose name matches, return the one with the highest 

293 name-similarity ratio (ties resolve to scores.json order). 

294 """ 

295 if lat is None or lon is None or not normalise_name(name): 

296 return None 

297 

298 target = " ".join(sorted(normalise_name(name).split())) 

299 best: dict | None = None 

300 best_ratio = -1.0 

301 for record in records: 

302 rlat, rlon = record.get("lat"), record.get("lon") 

303 if rlat is None or rlon is None: 

304 continue 

305 # Cheap bounding-box reject before the (relatively costly) name compare. 

306 if abs(rlat - lat) > GEO_RADIUS_DEG or abs(rlon - lon) > GEO_RADIUS_DEG: 

307 continue 

308 if not names_match(name, record.get("dba_name", "")): 

309 continue 

310 ratio = difflib.SequenceMatcher( 

311 None, target, " ".join(sorted(normalise_name(record.get("dba_name", "")).split())) 

312 ).ratio() 

313 if ratio > best_ratio: 

314 best_ratio = ratio 

315 best = record 

316 return best

319def name_search(query: str, records: list[dict], limit: int = 8) -> list[dict[str, Any]]: 

320 """Rank score records whose `dba_name` denotes the establishment `query` names. 

321 

322 A name-only resolver for the general-chat lookup: no address, no 

323 coordinates, just a name the user typed. Keeps every record that clears the 

324 same `names_match` identity rule the geo path uses, ranked by 

325 name-similarity (best first). Returns at most `limit` candidates. 

326 

327 The caller decides what to do with the shape of the result: 

328 * exactly one candidate -> a confident match, 

329 * several candidates -> a common name (a chain, or two venues with the 

330 same name) to disambiguate by address, 

331 * empty -> no city inspection record for that name (the 

332 caller must NOT invent one). 

333 

334 A blank / whitespace query, or one that normalises to nothing, returns [] 

335 rather than matching everything. 

336 """ 

337 if not normalise_name(query): 337 ↛ 338line 337 didn't jump to line 338 because the condition on line 337 was never true

338 return [] 

339 

340 target = " ".join(sorted(normalise_name(query).split())) 

341 scored: list[tuple[float, dict]] = [] 

342 for record in records: 

343 dba = record.get("dba_name", "") 

344 if not names_match(query, dba): 

345 continue 

346 ratio = difflib.SequenceMatcher( 

347 None, target, " ".join(sorted(normalise_name(dba).split())) 

348 ).ratio() 

349 scored.append((ratio, record)) 

350 

351 # Best match first; a stable sort keeps scores.json order among equal ratios 

352 # (same tie-break the address/geo paths use). 

353 scored.sort(key=lambda pair: pair[0], reverse=True) 

354 return [record for _ratio, record in scored[:limit]]

357def trend_label(slope: float | None) -> str: 

358 """Map a forecast trend slope to a plain-English direction label. 

359 

360 A null slope means the venue has fewer than 2 scored inspections, so no 

361 forward slope can be fit — that is "we can't say", not a flat trend. 

362 Reporting it as "stable" is what let the trend_slope_90d -> trend_slope 

363 rename miss (decision 0011) go silent in the deployed agent, so say so 

364 explicitly instead. 

365 """ 

366 if slope is None: 

367 return "not enough inspection history" 

368 if slope > 0.001: 

369 return "worsening" 

370 if slope < -0.001: 

371 return "improving" 

372 return "stable"

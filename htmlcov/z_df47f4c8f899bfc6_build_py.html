1"""Feature-table orchestrator. 

2 

3``build_features`` reads the contract inputs (the labeled inspections table 

4and the raw 311 complaints table), runs each feature module in turn, and 

5returns the modelling-ready DataFrame. 

6 

7Row filter applied at the END of the build, not at the start: 

8 * Burn-in (pre-2019) inspections are KEPT during feature computation so 

9 they can contribute to ``prior_*`` aggregates at later inspections. 

10 * They are DROPPED from the returned DataFrame because they can't be 

11 trained on (no label). 

12 * Rows with non-modelable results (Out of Business, No Entry, etc.) and 

13 placeholder license tokens are also dropped at the end. 

14 

15This separation between "keep for priors" and "use for training" is the 

16whole point of the burn-in design. If we filtered earlier we'd lose the 

17prior history at the start of 2019. 

18 

19See CLAUDE.md § "Code style" — every ``.shift()`` / ``< as_of_date`` guard 

20must be commented with WHY. The feature modules called from here document 

21their own leak guards. This orchestrator just stitches them together. 

22""" 

23 

24from __future__ import annotations 

25 

26import pandas as pd 

27 

28from foodsafety.features.building_features import add_building_features 

29from foodsafety.features.complaint_features import add_complaint_features 

30from foodsafety.features.inspection_features import add_inspection_features 

31from foodsafety.features.keyword_flags import add_keyword_flags 

32from foodsafety.features.license_features import add_license_features 

33from foodsafety.features.license_history_features import add_license_history_features 

34from foodsafety.features.temporal_features import add_temporal_features 

35from foodsafety.features.weather_features import add_weather_features 

36from foodsafety.utils.geo import warn_and_null_out_of_bbox 

37 

38MODELABLE_RESULTS = frozenset({"Pass", "Pass w/ Conditions", "Fail"}) 

39 

40 

41def build_features( 

42 inspections_labeled: pd.DataFrame, 

43 complaints: pd.DataFrame | None = None, 

44 licenses_historical: pd.DataFrame | None = None, 

45 building_permits: pd.DataFrame | None = None, 

46 building_violations: pd.DataFrame | None = None, 

47 weather: pd.DataFrame | None = None, 

48 *, 

49 drop_burnin: bool = True, 

50 drop_invalid_license: bool = True, 

51 drop_non_modelable: bool = True, 

52 drop_right_truncated: bool = False, 

53) -> pd.DataFrame: 

54 """End-to-end feature build. 

55 

56 Args: 

57 inspections_labeled: output of ``foodsafety.data.labels.build_labels``. 

58 Must include ``license_id``, ``inspection_date``, ``results``, 

59 ``violations``, ``is_burnin``, ``is_fail_or_priority``, label. 

60 complaints: 311 dataframe (``complaints_311.parquet``). If None, 

61 the complaint features are skipped — handy for tests. 

62 licenses_historical: optional Business Licenses history dataframe. 

63 If None, license-history features are skipped. 

64 weather: optional NOAA GHCN-Daily dataframe (``foodsafety.io.noaa. 

65 fetch_noaa_ghcnd`` output). If None, weather features are skipped. 

66 drop_burnin: drop rows with ``is_burnin=True`` from output. 

67 drop_invalid_license: drop rows with ``license_id`` in {"", "0"}. 

68 drop_non_modelable: drop rows whose ``results`` is not in 

69 ``MODELABLE_RESULTS``. 

70 drop_right_truncated: drop rows whose forward 180-day window extends 

71 past the dataset max date. Those labels are under-counted (we 

72 haven't observed the full window yet) so they bias eval metrics 

73 — but the FEATURES are valid for scoring. Default ``False`` so 

74 the parquet preserves them; the trainer filters them out before 

75 train/val/test, then scores against the full set. 

76 

77 Returns: 

78 New DataFrame. Original inputs not mutated. 

79 """ 

80 df = inspections_labeled.copy() 

81 df["inspection_date"] = pd.to_datetime(df["inspection_date"]) 

82 

83 # Build prior-history features FIRST. These use all rows including 

84 # burn-in (the whole reason burn-in exists is to feed `prior_*`). 

85 df = add_inspection_features(df) 

86 

87 # Static facility metadata — cheap and leak-safe. 

88 df = add_license_features(df) 

89 

90 # NLP layer B keyword flags. 

91 df = add_keyword_flags(df) 

92 

93 # Calendar / seasonality features (no external join needed). 

94 df = add_temporal_features(df) 

95 

96 # Spatial 311 counts (BallTree 300m). Skipped if no complaints provided. 

97 # Venue-level + neighborhood 311 angles (address-exact counts/recency/trend, 

98 # ring excess) were tested in Run 2 and dropped — all flat vs prior_* history 

99 # (see complaint_features.py + docs/model-experiments.md). The functions remain in 

100 # complaint_features.py, just not wired into the contract. 

101 if complaints is not None: 101 ↛ 102line 101 didn't jump to line 102 because the condition on line 101 was never true

102 df = add_complaint_features(df, complaints) 

103 

104 # License history (age + activity). Skipped if no historical-licenses df provided. 

105 if licenses_historical is not None: 105 ↛ 106line 105 didn't jump to line 106 because the condition on line 105 was never true

106 df = add_license_history_features(df, licenses_historical) 

107 

108 # Block-face building permits / violations (BallTree ~30m). Skipped if 

109 # neither building df is provided. Physical-plant condition, orthogonal to 

110 # inspection history; the leak guard (events strictly before the anchor) 

111 # lives in add_building_features. 

112 if building_permits is not None or building_violations is not None: 112 ↛ 113line 112 didn't jump to line 113 because the condition on line 112 was never true

113 df = add_building_features(df, building_permits, building_violations) 

114 

115 # Citywide daily weather (NOAA GHCN-Daily, O'Hare). Skipped if no weather 

116 # df is provided. Joined on date only, not grouped by license — see 

117 # weather_features.py for the leak guard. 

118 if weather is not None: 118 ↛ 119line 118 didn't jump to line 119 because the condition on line 118 was never true

119 df = add_weather_features(df, weather) 

120 

121 # Add as_of_date as a synonym of inspection_date — per contract the row 

122 # key is (license_id, as_of_date). We set them equal here so the contract 

123 # is honoured without changing semantics; the +1d convention discussed in 

124 # docs/interface_contracts.md is purely notational. 

125 df["as_of_date"] = df["inspection_date"] 

126 

127 # --- Output filter -------------------------------------------------- 

128 if drop_burnin: 128 ↛ 130line 128 didn't jump to line 130 because the condition on line 128 was always true

129 df = df[~df["is_burnin"]] 

130 if drop_invalid_license: 130 ↛ 132line 130 didn't jump to line 132 because the condition on line 130 was always true

131 df = df[~df["license_id"].isin({"", "0"})] 

132 if drop_non_modelable: 132 ↛ 134line 132 didn't jump to line 134 because the condition on line 132 was always true

133 df = df[df["results"].isin(MODELABLE_RESULTS)] 

134 if drop_right_truncated: 134 ↛ 141line 134 didn't jump to line 141 because the condition on line 134 was never true

135 # Anchors whose 180-day forward window extends past the dataset's 

136 # latest inspection date have under-counted labels: any future Fail 

137 # we haven't observed becomes a silent 0. Keep them out of training 

138 # AND eval — otherwise held-out metrics are biased (the test split is 

139 # the most affected since it's the most recent slice; on the current 

140 # snapshot about half of the test rows are right-truncated). 

141 df = df[~df["right_truncated"]] 

142 

143 # Geo sanity: warn + null (don't drop) any coords outside the Chicago bbox. 

144 # A bad geocode is a broken display field, not a reason to drop a valid 

145 # inspection — nulling routes it into the same missing-geo path the map 

146 # already handles (skip the pin), instead of plotting a pin in Lake Michigan. 

147 # Runs AFTER the 311 join, so it doesn't change the spatial counts on the 

148 # current snapshot (which is 100% in-bbox anyway); it's display + Phase-2 

149 # ingestion insurance. lat/lon aren't model features. 

150 df = warn_and_null_out_of_bbox(df) 

151 

152 return df.reset_index(drop=True)

1"""SHAP-style per-feature attribution for the linear baseline. 

2 

3For a calibrated logistic-regression pipeline, the "SHAP" decomposition has a 

4closed form: each feature's contribution is ``coef * scaled_value`` in 

5log-odds space. This is the exact local additive explanation (same as 

6``shap.LinearExplainer`` would produce for a linear model). We roll our own 

7in ~30 lines rather than pull the ``shap`` package — fewer dependencies, and 

8the math is uncomplicated. 

9 

10For the served XGBoost model, ``tree_contributions`` (below) uses XGBoost's 

11native TreeSHAP (``Booster.predict(pred_contribs=True)``) — same additive 

12log-odds decomposition, no ``shap`` package dependency. 

13 

14Two product surfaces depend on this module: 

15 

16 1. **The restaurant detail page** — top 3-5 drivers per restaurant, with 

17 plain-English labels. 

18 2. **The model card** — global feature impact (mean |contribution|), for the 

19 research write-up. 

20"""

82def linear_contributions( 

83 model, 

84 X: pd.DataFrame, 

85 original_features: list[str], 

86) -> pd.DataFrame: 

87 """Per-feature contributions in log-odds space for every row in ``X``. 

88 

89 Returns a DataFrame of shape ``(len(X), len(original_features))`` with the 

90 same row index as ``X`` and ``original_features`` as columns. For each 

91 cell, the value is the additive contribution of that ORIGINAL feature to 

92 the linear (log-odds) prediction. 

93 

94 For one-hot expanded categorical features (e.g., ``static_zip`` becomes 

95 60+ columns after the OneHotEncoder), we sum the contributions of all 

96 its expansions to attribute back to the original column. That keeps the 

97 per-feature view interpretable. 

98 """

101 raise TypeError(f"Expected sklearn Pipeline, got {type(pipeline).__name__}")

103 preprocess: ColumnTransformer = pipeline.named_steps["preprocess"] 

104 logreg: LogisticRegression = pipeline.named_steps["model"] 

105 

106 X_processed = preprocess.transform(X) 

107 if hasattr(X_processed, "toarray"): # sparse → dense (small N for prediction) 107 ↛ 108line 107 didn't jump to line 108 because the condition on line 107 was never true

108 X_processed = X_processed.toarray() 

109 coef = logreg.coef_[0] # binary classifier → 1D 

110 

111 # Element-wise: contribution per expanded column. 

112 expanded_contributions = X_processed * coef # (n_rows, n_expanded_cols) 

113

131def tree_contributions( 

132 xgb_estimator, 

133 X: pd.DataFrame, 

134 original_features: list[str], 

135) -> tuple[pd.DataFrame, float]: 

136 """Per-feature contributions in margin (log-odds) space for an XGB model. 

137 

138 Uses XGBoost's native TreeSHAP (``Booster.predict(pred_contribs=True)``) — 

139 the exact local additive explanation for the tree ensemble — so we avoid 

140 pulling the heavy ``shap`` package. The booster reports one contribution per 

141 input column plus a trailing bias term; for our XGB model each ORIGINAL 

142 feature is a single column (categoricals use native partition splits, not 

143 one-hot), so the contribution columns map 1:1 to ``original_features`` with 

144 no expansion to sum back. 

145 

146 The additive identity holds exactly: ``base_margin + Σ contributions == 

147 raw margin`` for every row, and the served Platt map turns that margin into 

148 the calibrated risk score. Returns ``(contrib_df, base_margin)`` where 

149 ``contrib_df`` is ``(len(X), len(original_features))`` and ``base_margin`` 

150 is the constant bias term (the model's expected margin) — shipped as the 

151 ``calibration.intercept`` so the app waterfall reconciles to the gauge. 

152 

153 ``X`` must already carry the dtypes the model trained on (categorical 

154 columns as ``category``); pass the prepared frame from the serve wrapper. 

155 """ 

156 import xgboost as xgb_lib 

157 

158 booster = xgb_estimator.get_booster() 

159 dmat = xgb_lib.DMatrix(X[original_features], enable_categorical=True) 

160 raw = booster.predict(dmat, pred_contribs=True) # (n, n_features + 1); last col = bias 

161 contrib_df = pd.DataFrame( 

162 raw[:, :-1], index=X.index, columns=list(original_features), dtype="float64" 

163 ) 

164 base_margin = float(raw[0, -1]) # constant across rows 

165 return contrib_df, base_margin

206def top_drivers_for_row( 

207 row_values: pd.Series, 

208 row_contributions: pd.Series, 

209 *, 

210 k: int = 4, 

211 labels: Mapping[str, str | Mapping[bool, str]] | None = None, 

212) -> list[Driver]: 

213 """Pick the top-K drivers for one restaurant. 

214 

215 "Top" means largest absolute log-odds contribution. We prefer **positive** 

216 contributions (features that push risk UP) but include negative ones if 

217 they're materially large, since "no prior fails" is a legitimate driver 

218 too. 

219 

220 Boolean flag features only render when they fired (True). A False flag 

221 contributes 0 to log-odds anyway, so it's correctly excluded by the 

222 magnitude sort, but this also keeps the labelling tidy if there's a 

223 rounding artefact. 

224 """ 

225 labels = labels or FEATURE_LABELS 

226 

227 # Hide boolean-False rows from drivers (their contribution is 0; suppress 

228 # any near-zero negative noise that might rank in the top-K). 

229 eligible_mask = [] 

230 for feat in row_contributions.index: 

231 if feat.startswith("flag_kw_"): 

232 eligible_mask.append(bool(row_values.get(feat, False))) 

233 else: 

234 eligible_mask.append(True) 

235 eligible = pd.Series(eligible_mask, index=row_contributions.index) 

236 

237 # Rank by magnitude over the eligible features. 

238 contribs = row_contributions.where(eligible, 0.0) 

239 ranked = contribs.reindex(contribs.abs().sort_values(ascending=False).index) 

240 top = ranked.head(k) 

241 

242 drivers: list[Driver] = [] 

243 for feat, shap_val in top.items(): 

244 if abs(shap_val) < 1e-6: 244 ↛ 245line 244 didn't jump to line 245 because the condition on line 244 was never true

245 continue # don't show purely-zero contributors 

246 raw_value = row_values.get(feat) 

247 label_template = labels.get(feat, feat) 

248 if isinstance(label_template, Mapping): 

249 # Sign-aware label for a binary outcome feature: the same feature 

250 # reads oppositely for the fail vs pass case (and surfaces in both). 

251 label = label_template.get(bool(raw_value), feat) 

252 else: 

253 try: 

254 label = label_template.format(value=raw_value) 

255 except (TypeError, ValueError): 

256 label = label_template 

257 drivers.append( 

258 Driver( 

259 feature=feat, 

260 value=str(raw_value), 

261 shap=float(shap_val), 

262 label=label, 

263 ) 

264 ) 

265 return drivers

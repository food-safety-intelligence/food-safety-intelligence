1"""Per-license inspection-history features (the workhorse `prior_*` columns). 

2 

3Every column added here is **leak-free** — at each anchor row, the feature 

4value summarises only data with strictly earlier dates at the same license. 

5The current row's own outcome never enters its own feature value. 

6 

7There are two leak-free patterns in this module and we use them consistently: 

8 

9 1. **Exclusive cumulative count.** 

10 ``group['x'].cumsum() - group['x']`` 

11 The cumsum INCLUDES the current row; subtracting the current row's 

12 contribution leaves the count over strictly earlier rows. 

13 

14 2. **Last-event-strictly-before.** 

15 Mask event values; forward-fill within group; shift by 1. 

16 The ffill gives the last event at-or-before this row; the shift moves 

17 it to "before this row" without including the row itself. 

18 

19If a teammate ever writes a `prior_*` feature without one of these two 

20patterns, ``tests/test_features.py`` should catch it (we assert leak-freeness 

21on synthetic data with a known answer). 

22 

23See CLAUDE.md § "Code style" — every ``.shift()`` / ``< as_of_date`` guard 

24must be commented with WHY. The comments below are mandatory, not optional. 

25""" 

26 

27from __future__ import annotations 

28 

29import numpy as np 

30import pandas as pd 

31 

32 

33def add_inspection_features( 

34 df: pd.DataFrame, 

35 *, 

36 license_col: str = "license_id", 

37 date_col: str = "inspection_date", 

38) -> pd.DataFrame: 

39 """Append per-anchor prior-history features. 

40 

41 Required input columns (from ``inspections_labeled.parquet``): 

42 - ``license_id``, ``inspection_date`` 

43 - ``is_fail_or_priority`` : bool — anchor event flag (from labels.py) 

44 - ``results`` : string — so we can isolate Fail anchors 

45 - ``violations`` : string — for the violation-code rollups 

46 

47 Returns a new DataFrame. Does not mutate the input. 

48 

49 Features added (every one of them describes events STRICTLY BEFORE the 

50 anchor's inspection_date at the same license_id): 

51 

52 prior_inspections int — count of prior inspections 

53 prior_fails int — count of prior Fail results 

54 prior_priority_violations int — count of prior priority (code 1-29) violations 

55 prior_core_violations int — count of prior core (code 30+) violations 

56 prior_fail_or_priority_events int — count of prior is_fail_or_priority=True rows 

57 prior_pass_w_conditions int — count of prior "Pass w/ Conditions" results 

58 prior_reinspections int — count of prior Re-Inspection visits 

59 prior_complaint_inspections int — count of prior Complaint-triggered visits 

60 prior_fail_rate float — prior_fails / prior_inspections, NaN if no priors 

61 prior_event_rate float — prior_fail_or_priority / prior_inspections, NaN if no priors 

62 days_since_last_inspection float — days; NaN if this is the first inspection 

63 days_since_last_fail float — days since last prior Fail; NaN if no prior Fail 

64 last_was_fail float — was the previous inspection a Fail (1/0); NaN if first 

65 prev_priority_violations float — priority violations at the previous inspection; NaN if first 

66 priority_violation_trend float — prev minus prev-prev priority count (worsening if >0) 

67 prior_fails_365d float — Fails in the trailing 365 days (leak-free, exclusive) 

68 prior_priority_violations_365d float — priority violations in trailing 365 days (leak-free) 

69 

70 Plus the anchor inspection's OWN outcome (current-inspection, not prior_*; 

71 leak-free because the label window is strictly AFTER the anchor): 

72 

73 was_fail int — was THIS inspection a Fail (1/0) 

74 n_priority_this_inspection int — priority (code 1-29) count on THIS inspection 

75 n_core_this_inspection int — core (code 30+) count on THIS inspection 

76 """ 

77 # We need stable sort order for the cumsum/shift patterns to produce 

78 # the right semantics. The group-key sort is also required for the 

79 # ffill-within-group dance below. 

80 out = df.sort_values([license_col, date_col], kind="mergesort").reset_index(drop=True) 

81 

82 # --- Building-block event-flag columns --------------------------------- 

83 # We use int columns rather than bool because cumsum on bool is allowed 

84 # but produces a confusing dtype. Int8 is plenty for these counts at row level. 

85 out["_is_fail_int"] = (out["results"] == "Fail").astype("int8") 

86 # Re-derive violation counts here rather than trusting upstream — keeps 

87 # this module self-contained and lets tests pass a minimal input. 

88 out["_priority_int"] = out["violations"].apply(_count_priority).astype("int32") 

89 out["_core_int"] = out["violations"].apply(_count_core).astype("int32") 

90 # is_fail_or_priority may already exist from labels.py — re-compute if not. 

91 if "is_fail_or_priority" not in out.columns: 91 ↛ 92line 91 didn't jump to line 92 because the condition on line 91 was never true

92 out["is_fail_or_priority"] = (out["_is_fail_int"] == 1) | (out["_priority_int"] > 0) 

93 out["_event_int"] = out["is_fail_or_priority"].astype("int8") 

94 

95 # Building blocks for the new prior_* rollups: near-misses ("Pass w/ 

96 # Conditions") and the visit trigger (Re-Inspection / Complaint). The 

97 # inspection_type column is absent in minimal test inputs → treat as 

98 # no-signal so the module stays self-contained. 

99 out["_pass_cond_int"] = (out["results"] == "Pass w/ Conditions").astype("int8") 

100 _itype = ( 

101 out["inspection_type"].astype("string").fillna("").str.lower() 

102 if "inspection_type" in out.columns 

103 else pd.Series("", index=out.index) 

104 ) 

105 out["_reinsp_int"] = _itype.str.contains("re-inspection|reinspection", regex=True).astype( 

106 "int8" 

107 ) 

108 out["_complaint_int"] = _itype.str.contains("complaint", regex=True).astype("int8") 

109 

110 group = out.groupby(license_col, sort=False) 

111 

112 # --- Pattern 1: exclusive cumulative counts ---------------------------- 

113 # cumsum() includes the current row. Subtracting the row's value gives 

114 # the count over strictly earlier rows at this license. This is the 

115 # leak-free guard. Do NOT replace these subtractions with .shift() — 

116 # shift() is not group-aware and would bleed values across licenses at 

117 # group boundaries. 

118 out["prior_inspections"] = ( 

119 group.cumcount().astype("int32") 

120 # cumcount() is 0-indexed and naturally excludes the current row already 

121 ) 

122 out["prior_fails"] = (group["_is_fail_int"].cumsum() - out["_is_fail_int"]).astype("int32") 

123 out["prior_priority_violations"] = ( 

124 group["_priority_int"].cumsum() - out["_priority_int"] 

125 ).astype("int32") 

126 out["prior_core_violations"] = (group["_core_int"].cumsum() - out["_core_int"]).astype("int32") 

127 out["prior_fail_or_priority_events"] = ( 

128 group["_event_int"].cumsum() - out["_event_int"] 

129 ).astype("int32") 

130 # Prior near-misses and prior visit-trigger mix — same exclusive-cumsum 

131 # guard (count over strictly-earlier rows at this license). 

132 out["prior_pass_w_conditions"] = ( 

133 group["_pass_cond_int"].cumsum() - out["_pass_cond_int"] 

134 ).astype("int32") 

135 out["prior_reinspections"] = (group["_reinsp_int"].cumsum() - out["_reinsp_int"]).astype( 

136 "int32" 

137 ) 

138 out["prior_complaint_inspections"] = ( 

139 group["_complaint_int"].cumsum() - out["_complaint_int"] 

140 ).astype("int32") 

141 

142 # Ratios: only defined when there's at least one prior. NaN otherwise — 

143 # tree models handle NaN natively; logreg pipelines should impute. 

144 safe_denom = out["prior_inspections"].replace(0, np.nan) 

145 out["prior_fail_rate"] = (out["prior_fails"] / safe_denom).astype("float64") 

146 out["prior_event_rate"] = (out["prior_fail_or_priority_events"] / safe_denom).astype("float64") 

147 

148 # --- Pattern 2: "last event strictly before" -------------------------- 

149 # days_since_last_inspection: shift the date within the license group. 

150 # ``group_keys=False`` keeps the result aligned with the original index. 

151 last_insp = group[date_col].shift(1) 

152 out["days_since_last_inspection"] = (out[date_col] - last_insp).dt.days.astype("float64") 

153 

154 # days_since_last_fail: more involved. We want the date of the most 

155 # recent Fail at this license BEFORE the current row. 

156 # 

157 # The trick: 

158 # 1. Build a Series whose value is the date when results=='Fail', NaT otherwise. 

159 # 2. ffill WITHIN the license group — each row now carries the date of 

160 # the most recent Fail at-or-before this row. 

161 # 3. shift(1) WITHIN the license group — now each row carries the date 

162 # of the most recent Fail STRICTLY before this row. The shift also 

163 # correctly leaves NaT in the first row of each group. 

164 # 

165 # Step 2's groupby().ffill() and Step 3's groupby().shift() each respect 

166 # group boundaries, so there's no cross-license leak. 

167 fail_date_or_nat = out[date_col].where(out["_is_fail_int"] == 1, pd.NaT) 

168 last_fail_at_or_before = fail_date_or_nat.groupby(out[license_col], sort=False).ffill() 

169 last_fail_strictly_before = last_fail_at_or_before.groupby(out[license_col], sort=False).shift( 

170 1 

171 ) 

172 out["days_since_last_fail"] = (out[date_col] - last_fail_strictly_before).dt.days.astype( 

173 "float64" 

174 ) 

175 

176 # --- Recency / last-outcome / trend (own-history only; leak-free) ------ 

177 # These describe the restaurant's OWN prior inspections — individually fair, 

178 # with no neighborhood/demographic proxy. (We deliberately DECLINED a 

179 # neighborhood peer-fail feature on fairness grounds; see docs/weekly.) 

180 # shift(1)/shift(2) reference strictly-earlier inspections within the 

181 # license group; the 365d window counts only earlier rows. 

182 out["last_was_fail"] = group["_is_fail_int"].shift(1).astype("float64") 

183 out["prev_priority_violations"] = group["_priority_int"].shift(1).astype("float64") 

184 # Trend: priority-violation count at the previous inspection minus the one 

185 # before that. Positive ⇒ the last visit was worse than the prior visit 

186 # (deteriorating). NaN until there are two priors. 

187 out["priority_violation_trend"] = ( 

188 group["_priority_int"].shift(1) - group["_priority_int"].shift(2) 

189 ).astype("float64") 

190 # Trailing-365d counts: a recency-weighted alternative to the LIFETIME 

191 # prior_fails / prior_priority_violations totals. Tests whether recent 

192 # history beats all-time history on a non-stationary risk process. 

193 out["prior_fails_365d"] = _trailing_window_count( 

194 out, license_col, date_col, "_is_fail_int", 365 

195 ) 

196 out["prior_priority_violations_365d"] = _trailing_window_count( 

197 out, license_col, date_col, "_priority_int", 365 

198 ) 

199 

200 # --- Current-inspection own outcome (NOT a prior_* feature) ------------- 

201 # The anchor inspection's OWN result and code counts. Leak-free WITHOUT a 

202 # shift: these are observed AT as_of_date (== inspection_date), and the 

203 # 180-day label window is strictly AFTER the anchor — so the anchor's own 

204 # outcome can't leak its own forward label. The model otherwise sees only 

205 # PRIOR outcomes (the shift/cumsum prior_* columns) and the current 

206 # comment's keyword flags — never the current visit's own Fail result or 

207 # violation-code counts. These are promoted, not shifted, on purpose. 

208 # 

209 # Re-inspection dynamic to keep in mind (validated at eval, not here): a 

210 # Fail triggers a mandated ~30-day re-inspection that often lands inside 

211 # the 180-day window. That's legitimate forward signal, but it means 

212 # was_fail correlates with the label partly via that mechanism — see 

213 # docs/model-experiments.md for the circularity check. 

214 out["was_fail"] = out["_is_fail_int"].astype("int8") 

215 out["n_priority_this_inspection"] = out["_priority_int"].astype("int32") 

216 out["n_core_this_inspection"] = out["_core_int"].astype("int32") 

217 

218 # Drop intermediate columns; they were named with `_` prefix on purpose. 

219 return out.drop( 

220 columns=[ 

221 "_is_fail_int", 

222 "_priority_int", 

223 "_core_int", 

224 "_event_int", 

225 "_pass_cond_int", 

226 "_reinsp_int", 

227 "_complaint_int", 

228 ] 

229 ) 

230 

231

233# Helpers — kept local to this module so the file is self-contained. 

234# We import from `labels.py` would also work but couples the modules; the 

235# duplication is two lines and trivially correct.

238import re as _re # noqa: E402 (deliberate: keep this helper self-contained at file end) 

239 

240_VIOLATION_CODE_RE = _re.compile(r"(?:^|\|)\s*(\d{1,2})\.\s") 

241 

242 

243def _violation_codes(text: object) -> list[int]: 

244 if not isinstance(text, str): 

245 return [] 

246 return [int(c) for c in _VIOLATION_CODE_RE.findall(text)] 

247 

248 

249def _count_priority(text: object) -> int: 

250 return sum(1 for c in _violation_codes(text) if c <= 29) 

251 

252 

253def _count_core(text: object) -> int: 

254 return sum(1 for c in _violation_codes(text) if c > 29)

257def _trailing_window_count( 

258 out: pd.DataFrame, 

259 license_col: str, 

260 date_col: str, 

261 value_col: str, 

262 window_days: int, 

263) -> np.ndarray: 

264 """Per-anchor sum of ``value_col`` over the restaurant's own STRICTLY 

265 EARLIER inspections within ``window_days`` (leak-free; current row excluded). 

266 

267 Requires ``out`` sorted by ``[license_col, date_col]`` (the caller sorts 

268 once at the top of ``add_inspection_features``), so each license's rows are 

269 a contiguous, date-ascending block. Within a block we prefix-sum the values 

270 and use ``searchsorted`` to find ``lo`` — the first earlier row still inside 

271 the window — then ``cum[i] - cum[lo]`` sums rows ``[lo, i)``: earlier rows 

272 only, current row ``i`` excluded. This is the same leak-free guarantee as 

273 the exclusive-cumsum ``prior_*`` columns, but bounded to a time window. 

274 """ 

275 dates = out[date_col].to_numpy().astype("datetime64[D]") 

276 vals = out[value_col].to_numpy() 

277 lic = out[license_col].to_numpy() 

278 res = np.zeros(len(out), dtype="float64") 

279 win = np.timedelta64(window_days, "D") 

280 # Contiguous license blocks (out is sorted by [license, date]). 

281 bounds = np.flatnonzero(np.r_[True, lic[1:] != lic[:-1], True]) 

282 for s, e in zip(bounds[:-1], bounds[1:], strict=True): 

283 d = dates[s:e] 

284 cum = np.concatenate(([0], np.cumsum(vals[s:e]))) 

285 lo = np.searchsorted(d, d - win, side="right") 

286 idx = np.arange(e - s) 

287 res[s:e] = cum[idx] - cum[lo] 

288 return res

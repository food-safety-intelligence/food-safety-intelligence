1"""Calendar-based features derived from ``inspection_date``. 

2 

3Cheap, leak-free, and well-motivated: 

4 

5 * **Seasonality.** Food-safety violations are seasonal — Chicago summers 

6 drive cold-holding failures, winters the inverse. The 2014 Chicago 

7 forecasting paper used a 3-day rolling high temperature; in lieu of NOAA 

8 (deferred per CLAUDE.md), month and quarter are zero-cost proxies. 

9 * **Day-of-week.** Inspection cadence varies by weekday. Weekend 

10 inspections are rare; Monday inspections often follow weekend complaints. 

11 

12No leak risk: these are derived purely from the anchor's own date, no 

13dependence on any other row. 

14""" 

15 

16from __future__ import annotations 

17 

18import pandas as pd 

19 

20# Quarter → season mapping used by the inspection-procedure literature. 

21# Winter = Dec/Jan/Feb (months 12/1/2), Spring = Mar/Apr/May, etc. We treat 

22# December as Winter — a calendar quarter would put it with Q4. 

23_MONTH_TO_SEASON = { 

24 12: "Winter", 

25 1: "Winter", 

26 2: "Winter", 

27 3: "Spring", 

28 4: "Spring", 

29 5: "Spring", 

30 6: "Summer", 

31 7: "Summer", 

32 8: "Summer", 

33 9: "Fall", 

34 10: "Fall", 

35 11: "Fall", 

36} 

37 

38 

39def add_temporal_features( 

40 df: pd.DataFrame, 

41 *, 

42 date_col: str = "inspection_date", 

43) -> pd.DataFrame: 

44 """Append calendar features derived from ``date_col``. 

45 

46 Features added: 

47 temporal_month (Int8) — 1..12 

48 temporal_dow (Int8) — 0=Monday .. 6=Sunday 

49 temporal_quarter (Int8) — 1..4 

50 temporal_year (Int16) — calendar year (for year-fixed-effects) 

51 temporal_season (category) — Winter / Spring / Summer / Fall 

52 """ 

53 out = df.copy() 

54 out[date_col] = pd.to_datetime(out[date_col]) 

55 

56 out["temporal_month"] = out[date_col].dt.month.astype("Int8") 

57 out["temporal_dow"] = out[date_col].dt.dayofweek.astype("Int8") 

58 out["temporal_quarter"] = out[date_col].dt.quarter.astype("Int8") 

59 out["temporal_year"] = out[date_col].dt.year.astype("Int16") 

60 out["temporal_season"] = ( 

61 out["temporal_month"].astype(int).map(_MONTH_TO_SEASON).astype("category") 

62 ) 

63 

64 return out

67def override_scoring_month(df: pd.DataFrame, *, month: int) -> pd.DataFrame: 

68 """Freeze the seasonal calendar features to one common month for SCORING. 

69 

70 Seasonality is a real population effect (late-year anchors have higher 

71 forward-180d risk because the window runs into warm season), so we keep 

72 `temporal_month` / `temporal_quarter` in the trained model. But the web app 

73 anchors each venue at its MOST RECENT inspection and would otherwise freeze 

74 the seasonal term at that venue's inspection month -- making otherwise 

75 identical venues differ purely by inspection timing and presenting a stale 

76 season (e.g. a venue last seen in December) as timeless risk. So at SCORING 

77 time only -- never on train/val/test, which must evaluate on real months -- 

78 we override both seasonal columns to one common `month` (the as-of month). 

79 

80 Only `temporal_month` and `temporal_quarter` feed the model; `temporal_season` 

81 / `temporal_year` / `temporal_dow` are excluded from the feature set, so those 

82 two columns are all that need freezing. Returns a copy; input is untouched. 

83 """ 

84 if not 1 <= month <= 12: 

85 raise ValueError(f"month must be 1..12, got {month}") 

86 out = df.copy() 

87 quarter = (month - 1) // 3 + 1 

88 # Preserve the Int8 dtype add_temporal_features produced so downstream 

89 # prepare_xgb_features sees an unchanged schema. 

90 out["temporal_month"] = pd.Series(month, index=out.index, dtype="Int8") 

91 out["temporal_quarter"] = pd.Series(quarter, index=out.index, dtype="Int8") 

92 return out

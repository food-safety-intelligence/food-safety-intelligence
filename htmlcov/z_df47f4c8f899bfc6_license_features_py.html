19def normalize_facility_type(raw: str | float | None) -> str | None: 

20 """Collapse the messy raw ``facility_type`` into a canonical bucket. 

21 

22 Chicago's ``facility_type`` is operator-entered free text — ~500 distinct 

23 values with heavy casing/spelling fragmentation (the Daycare family alone 

24 spans ~20 spellings: "DAYCARE", "Daycare (2 - 6 Years)", "Day Care 1023", …). 

25 This consolidates the **vulnerable-population** families so the 

26 group-performance fairness audit (``notebooks/06``) has statistically 

27 meaningful buckets instead of a long fragmented tail. 

28 

29 AUDIT / UI ONLY — **not a model feature.** ``static_facility_type`` is dropped 

30 from the model for fairness (see decision 0004), so this never feeds a 

31 prediction; it only sharpens the audit's per-group view (and any future UI 

32 display). Returns ``None`` for missing / blank input. 

33 

34 Rule ORDER matters: senior / long-term care is matched BEFORE the daycare 

35 rule so "adult daycare" / "senior day care" land in Long Term Care, not the 

36 child Daycare bucket; adult culinary schools are matched before the child 

37 School rule for the same reason. 

38 """ 

39 # Missing in any form: Python None, float NaN, or pandas pd.NA. The audit calls 

40 # this on a StringDtype column, where missing is pd.NA (not float NaN), so a 

41 # float-only check would let it through and title-case it into a "<Na>" bucket. 

42 if raw is None or raw is pd.NA or (isinstance(raw, float) and pd.isna(raw)): 

43 return None 

44 s = re.sub(r"\s+", " ", str(raw).strip()) 

45 if not s: 

46 return None 

47 low = s.lower() 

48 # Senior / long-term (adult) care FIRST — keeps "adult daycare" out of Daycare. 

49 if any( 

50 k in low 

51 for k in ("long term care", "nursing home", "assisted living", "senior day", "adult day") 

52 ): 

53 return "Long Term Care" 

54 if "daycare" in low or "day care" in low: 

55 return "Daycare" 

56 if "child" in low and ("service" in low or "activity" in low): 

57 return "Children's Services Facility" 

58 # Adult culinary schools are NOT child schools. 

59 if any(k in low for k in ("culinary", "cooking school", "pastry school", "teaching school")): 

60 return "Culinary School" 

61 if "school" in low: 

62 return "School" 

63 if "shelter" in low and "animal" not in low: 

64 return "Shelter" 

65 if "hospital" in low and "restaurant" not in low: 65 ↛ 66line 65 didn't jump to line 66 because the condition on line 65 was never true

66 return "Hospital" 

67 # Long tail: title-case so casing-only duplicates (TAVERN/Tavern) merge. The 

68 # remaining tail is tiny per value and excluded by the audit's n>=50 gate. 

69 return s.title()

72def add_license_features(df: pd.DataFrame) -> pd.DataFrame: 

73 """Append static categorical features. 

74 

75 Features added: 

76 static_facility_type — Restaurant / Grocery Store / Bakery / etc. 

77 static_risk_tier — Chicago's pre-assigned tier ("Risk 1 (High)" / 2 / 3) 

78 static_zip — 5-digit ZIP 

79 static_zip3 — first 3 digits of ZIP. Less sparse than full 

80 ZIP; useful as a fallback when ZIP has few rows. 

81 static_inspection_type — scheduled inspection trigger (Canvass / 

82 Complaint / Re-Inspection / License). Known at 

83 scheduling time, before the outcome. 

84 

85 All four are cast to pandas' nullable ``category`` dtype — XGBoost reads 

86 these natively, and scikit-learn pipelines can one-hot via 

87 ``OrdinalEncoder`` / ``OneHotEncoder``. 

88 """ 

89 out = df.copy() 

90 

91 out["static_facility_type"] = out["facility_type"].astype("string").astype("category") 

92 out["static_risk_tier"] = out["risk"].astype("string").astype("category") 

93 

94 # The scheduled inspection trigger (Canvass / Complaint / Re-Inspection / 

95 # License). It's known when the visit is dispatched — before the outcome — 

96 # so it's leak-safe as an anchor categorical, same status as facility_type. 

97 # Absent in minimal test inputs → fall back to "Unknown". 

98 if "inspection_type" in out.columns: 98 ↛ 99line 98 didn't jump to line 99 because the condition on line 98 was never true

99 out["static_inspection_type"] = ( 

100 out["inspection_type"].astype("string").fillna("Unknown").astype("category") 

101 ) 

102 else: 

103 out["static_inspection_type"] = pd.Series("Unknown", index=out.index, dtype="category") 

104 

105 # ZIP cleaning: cast to string, strip trailing decimals (some upstream 

106 # parquet round-trips can yield "60614.0"), then require an exact 5-digit 

107 # match. Short codes ("606") and missing values both become "" — we do NOT 

108 # zero-pad short codes, because "00606" isn't a real Chicago ZIP and the 

109 # padding would silently invent a value that doesn't exist. 

110 zip_clean = out["zip"].astype("string").fillna("").str.replace(r"\.0$", "", regex=True) 

111 zip_clean = zip_clean.where(zip_clean.str.match(r"^\d{5}$"), "") 

112 out["static_zip"] = zip_clean.astype("category") 

113 out["static_zip3"] = zip_clean.str.slice(0, 3).astype("category") 

114 

115 return out

1"""Layer-C (dense) NLP — violation free-text embeddings as model features. 

2 

3The hybrid violation-text strategy in CLAUDE.md is A (structured code counts), 

4B (hand-picked keyword flags), C (dense text). This module is the *dense* arm 

5of C: instead of the flat TF-IDF -> TruncatedSVD that came up flat, it joins 

6**precomputed sentence-embedding columns** onto the feature frame. The 

7embeddings themselves are produced offline by ``scripts/build_text_embeddings.py`` 

8(a one-time Bedrock batch pass, cached to parquet) — never at request time and 

9never inside this module. This is the permanent batch -> cache -> features 

10pattern; nothing here calls a model. 

11 

12Join key is the **text itself** (a content hash), not the inspection key: the 

13embedding is a pure function of the violation comment, and ``(license_id, 

14as_of_date)`` is NOT unique (a licence can have two inspections on one day with 

15different text). Hashing the text dedupes natively and matches each row to the 

16embedding of its OWN comment. 

17 

18Leak-free contract (the WHY, per CLAUDE.md): 

19 Each row receives the embedding of its OWN current violations text, which is 

20 observed at ``as_of_date`` — the exact same justification as the ``was_fail`` 

21 / ``n_priority_this_inspection`` current-inspection features (label window is 

22 strictly AFTER as_of_date). The join is a pure per-text lookup: NO cross-row 

23 aggregation, NO prior-mean embedding, NO ``.shift()``. A prior-mean embedding 

24 WOULD need the exclusive-window guard the prior_* features use, so we do not 

25 build one here. 

26""" 

27 

28from __future__ import annotations 

29 

30import hashlib

32import pandas as pd 

33 

34# Cache column convention. The offline builder writes one float column per 

35# embedding dimension as ``txt_emb_000``..``txt_emb_NNN`` plus the ``text_hash`` 

36# join key. Reduction (PCA fit on TRAIN only) happens at modelling time, not here 

37# — a train-fit transform can't be precomputed without leaking the split. 

38EMBED_PREFIX: str = "txt_emb_" 

39HAS_TEXT_COL: str = "has_violation_text" 

40TEXT_HASH_COL: str = "text_hash" 

41 

42 

43def embedding_columns(df: pd.DataFrame) -> list[str]: 

44 """The ``txt_emb_*`` columns present in ``df``, in sorted (dimension) order.""" 

45 return sorted(c for c in df.columns if c.startswith(EMBED_PREFIX)) 

46 

47 

48def violation_text_hash(violations: pd.Series) -> pd.Series: 

49 """Stable content hash of each row's normalised violations text. 

50 

51 Normalisation (string-cast -> fill missing -> strip) MUST match the offline 

52 builder so the hashes line up. Rows with empty/missing text hash to ``<NA>`` 

53 so they don't join to any embedding (they get the zero vector instead). 

54 """ 

55 norm = violations.astype("string").fillna("").str.strip() 

56 return norm.map(lambda s: hashlib.sha1(s.encode("utf-8")).hexdigest() if s else pd.NA).astype( 

57 "string" 

58 )

61def add_text_embedding_features( 

62 df: pd.DataFrame, 

63 embeddings: pd.DataFrame, 

64) -> pd.DataFrame: 

65 """Left-join the cached violation-text embedding columns by text content. 

66 

67 Args: 

68 df: feature frame; must contain the ``violations`` text column. 

69 embeddings: cache from ``scripts/build_text_embeddings.py`` — one row per 

70 distinct comment, with ``text_hash`` plus ``txt_emb_*`` columns. 

71 

72 Returns: 

73 New frame = ``df`` plus the ``txt_emb_*`` columns and a 

74 ``has_violation_text`` 0/1 flag. Rows with empty / missing violations 

75 text (or text not in the cache) get a **zero vector** and 

76 ``has_violation_text=0`` — the flag lets the model separate "text says 

77 nothing risky" from "there was no text". 

78 

79 Leak-free: a pure left join on the row's OWN text hash. The output for a row 

80 depends only on that row's own comment; row order does not matter and no 

81 other row's text is consulted. 

82 """ 

83 emb_cols = embedding_columns(embeddings) 

84 if not emb_cols: 84 ↛ 85line 84 didn't jump to line 85 because the condition on line 84 was never true

85 raise ValueError("embeddings frame has no txt_emb_* columns") 

86 

87 out = df.copy() 

88 out[TEXT_HASH_COL] = violation_text_hash(out["violations"]) 

89 cache = embeddings[[TEXT_HASH_COL, *emb_cols]].copy() 

90 cache[TEXT_HASH_COL] = cache[TEXT_HASH_COL].astype("string") 

91 # Defend against an accidentally-duplicated cache; the embedding is identical 

92 # for a given hash, so keeping the first is lossless and keeps the join 1:m. 

93 cache = cache.drop_duplicates(TEXT_HASH_COL) 

94 

95 out = out.merge(cache, on=TEXT_HASH_COL, how="left", validate="many_to_one") 

96 out[HAS_TEXT_COL] = out[emb_cols[0]].notna().astype("int8") 

97 out[emb_cols] = out[emb_cols].fillna(0.0) 

98 return out.drop(columns=[TEXT_HASH_COL])

1""" 

2Lambda handler: find_inspection_records 

3--------------------------------------- 

4Builds a deep link to the ACTIVE CITY's AUTHORITATIVE food-inspection records for a 

5SET of establishments the agent is discussing — a comparison, a short list, or an 

6area. This is the city's own inspection data (the source the risk model is built 

7from), so unlike find_reviews it carries no disclaimer: it is provenance the user 

8can verify, not third-party opinion. 

9 

10Same posture as find_reviews — we only BUILD a URL to the city's open-data query 

11view. No fetch, no credentials, nothing stored; the user clicks through. The link 

12opens the portal's query grid, filtered one of three ways by whichever input the 

13agent supplies: 

14 * license_ids -> each establishment's full inspection history 

15 * zip -> every inspection in that ZIP 

16 * lat + lon + radius_m -> every inspection within that radius 

17 

18The agent has license_ids for anything it has scored (its universal key — it holds 

19each city's native id: Chicago license number, NYC CAMIS); it has no inspection_id 

20(no tool exposes one) and dba_name matching is unreliable, so those are NOT inputs. 

21 

22City coverage: Chicago and NYC publish on Socrata, so a filtered query link works 

23for both. LA left Socrata for a bulk CSV with no queryable API, so there is no 

24filtered grid to link to — for LA the tool returns LA County Public Health's 

25inspections page (a lookup landing page), not a pre-filtered link. 

26"""

108def handler(event: dict[str, Any], _ctx: Any) -> dict[str, Any]: 

109 """ 

110 Input event — exactly ONE filter mode: 

111 {"license_ids": list[str]} # compare / list named places 

112 {"zip": str} # area by ZIP 

113 {"lat": float, "lon": float, "radius_m": float} # area by radius 

114 

115 Returns {url, mode, truncated, note} — or {error, reason} if no mode is given. 

116 license_ids takes precedence if more than one mode is supplied. `city` (from 

117 the entrypoint) selects the portal; defaults to Chicago. 

118 """ 

119 city = (event.get("city") or "chicago").strip().lower() 

120 portal = _CITY_PORTALS.get(city) 

121 if portal is None: 

122 # LA (and any future non-Socrata city): no queryable grid to filter. Hand 

123 # back the city's inspections landing page so the user still gets a real, 

124 # authoritative link to look records up themselves. 

125 if city == "la": 125 ↛ 136line 125 didn't jump to line 136 because the condition on line 125 was always true

126 return { 

127 "url": _LA_INSPECTIONS_URL, 

128 "mode": "city_page", 

129 "truncated": False, 

130 "note": ( 

131 "LA County's inspection data isn't a filtered link, so this opens " 

132 "LA County Public Health's restaurant & market inspections page, " 

133 "where the user can look a place up." 

134 ), 

135 } 

136 return { 

137 "error": f"find_inspection_records does not cover {city!r}.", 

138 "reason": "unsupported_city", 

139 } 

140 

141 license_ids = _dedupe_ids(event.get("license_ids")) 

142 zip_code = str(event.get("zip") or "").strip() 

143 lat, lon, radius_m = event.get("lat"), event.get("lon"), event.get("radius_m") 

144 

145 truncated = False 

146 if license_ids: 

147 shown = license_ids[:MAX_IDS] 

148 truncated = len(license_ids) > MAX_IDS 

149 where = _in_clause(portal["id_col"], shown) 

150 mode = "license_ids" 

151 elif zip_code: 

152 where = f"`{portal['zip_col']}`='{_soql_literal(zip_code)}'" 

153 mode = "zip" 

154 elif _is_number(lat) and _is_number(lon) and _is_number(radius_m): 

155 coords = f"{_fmt_num(lat)}, {_fmt_num(lon)}, {_fmt_num(radius_m)}" 

156 where = f"within_circle(`{portal['geo_col']}`, {coords})" 

157 mode = "geo" 

158 else: 

159 return { 

160 "error": ( 

161 "find_inspection_records needs one filter: license_ids, zip, or lat+lon+radius_m." 

162 ), 

163 "reason": "missing_filter", 

164 } 

165 

166 note = ( 

167 f"Opens the {portal['portal_name']} portal filtered to these records — the " 

168 "city's own data behind the risk score. It can take a few seconds to load." 

169 ) 

170 if truncated: 

171 note += f" Shows the first {MAX_IDS} of {len(license_ids)} establishments." 

172 

173 return {"url": _build_url(where, portal), "mode": mode, "truncated": truncated, "note": note}

176def _dedupe_ids(raw: Any) -> list[str]: 

177 """Normalise ids to a de-duplicated list of non-empty strings; order preserved. 

178 

179 A bare scalar (the model emitting one id as a number/string instead of a list) 

180 is treated as a single id, not iterated — so a malformed float can't crash the 

181 tool; None / a mapping yields no ids so the caller degrades to a filter error. 

182 """ 

183 if raw is None or isinstance(raw, dict): 

184 return [] 

185 if not isinstance(raw, (list, tuple)): 

186 raw = [raw] 

187 seen: list[str] = [] 

188 for item in raw: 

189 s = str(item).strip() 

190 if s and s not in seen: 

191 seen.append(s) 

192 return seen

195def _is_number(x: Any) -> bool: 

196 # bool is an int subclass — exclude it so True/False can't act as coordinates. 

197 return isinstance(x, (int, float)) and not isinstance(x, bool)

200def _fmt_num(x: float) -> str: 

201 """Compact numeric literal — drop a trailing '.0' so an integer radius reads 

202 as `300`, not `300.0` (matches the portal query-builder's own form).""" 

203 f = float(x) 

204 return str(int(f)) if f.is_integer() else repr(f)

207def _soql_literal(s: str) -> str: 

208 """Escape a SoQL single-quoted string literal ('' escapes an embedded quote).""" 

209 return s.replace("'", "''")

212def _in_clause(column: str, ids: list[str]) -> str: 

213 # The portal's query builder quotes IN values with double quotes; match that 

214 # form (verified to render). Ids are license numbers; strip any stray double 

215 # quote or backslash defensively so a value can't break out of the string 

216 # regardless of how the SoQL parser treats escapes. 

217 values = ", ".join(f'"{i.replace(chr(34), "").replace(chr(92), "")}"' for i in ids) 

218 return f"`{column}` IN ({values})"

221def _build_url(where: str, portal: dict[str, Any]) -> str: 

222 # Mirror the portal query-builder's own URL shape: an explicit newline-joined 

223 # column SELECT, the WHERE, then ORDER BY, percent-encoded into the path. 

224 select = "SELECT\n " + ",\n ".join(f"`{c}`" for c in portal["columns"]) 

225 soql = f"{select}\nWHERE {where}\nORDER BY `inspection_date` DESC NULL FIRST" 

226 return f"{portal['base']}/{urllib.parse.quote(soql, safe='')}/page/filter"
